# LearnGoLangProgram
Installation:

version:
  # go version

Go working
 1. # go install
 // Compile and install the copy of binary(named as module name) to the go/bin/. directory
   # binaryfilename
// To run the binary executable, located in the GOBIN path

 2. # go build
 //compile and install the copy in the working directory(go.mod location), we can change the path  

 3. # go run {main.go}
 // compile and install in a temporary location, go run --work main.go
